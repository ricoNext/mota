# 魔塔 MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 落地可本地运行的魔塔 Web MVP：纯 TS 核心 + Canvas 适配器，样板 3 层，含移动、钥匙门、战斗、商店、localStorage 存读档。

**Architecture:** `core/` 纯逻辑（无 DOM）处理状态步进与序列化；`data/easy/` 为 JSON 数据包；`adapters/web/` 用 Vite + Canvas + HTML 面板发命令并重绘。主循环：输入 → `step(state, command, pack)` → 新状态 → 渲染。日后 `adapters/wechat` 与 `data/hard` 复用同一 core。

**Tech Stack:** TypeScript、Vite、Vitest、Canvas 2D、localStorage、原生 ES modules（无 UI 框架）。

## Global Constraints

- 引擎：TypeScript 纯逻辑核心 + Canvas 薄渲染层；第一版不做 Taro / uni-app / 微信 adapter。
- 战斗：无经验值、无随机数；伤害由攻防差回合结算；打不过不可踩上怪物格。
- 公式锁定：`forecastDamage` 若 `hero.atk <= m.def` 返回 `null`；否则 `hits = ceil(m.hp / (hero.atk - m.def))`，`damage = max(0, m.atk - hero.def) * (hits - 1)`；可战条件为 `damage !== null && hero.hp > damage`；战后 `hp -= damage`。
- 存档：序列化英雄 + 各层可变 tiles；网页 `localStorage`；槽位 1～3。
- 第一版不做：地图编辑器 GUI、联机、必选音效、完整 21/50 关卡、微信登录。
- 标识符与文件名用英文；计划叙述可用中文。
- 非法行动返回未变更语义上的原状态副本 + `ReasonCode`，不抛异常中断循环。
- 存读档快捷键：`Ctrl+1/2/3` 存档，`Alt+1/2/3` 读档（避免与商店购买数字键冲突）。

import type { Fighter } from './types';

export function forecastDamage(hero: Fighter, monster: Fighter): number | null {
  if (hero.atk <= monster.def) return null;
  const hits = Math.ceil(monster.hp / (hero.atk - monster.def));
  const per = Math.max(0, monster.atk - hero.def);
  return per * (hits - 1);
}

export function canDefeat(hero: Fighter, monster: Fighter): boolean {
  const damage = forecastDamage(hero, monster);
  return damage !== null && hero.hp > damage;
}

import type { Dir, FloorState, GameState, Tile } from './types';

const DELTA: Record<Dir, { dx: number; dy: number }> = {
  up: { dx: 0, dy: -1 },
  down: { dx: 0, dy: 1 },
  left: { dx: -1, dy: 0 },
  right: { dx: 1, dy: 0 },
};

export function neighbor(x: number, y: number, dir: Dir): { x: number; y: number } {
  const { dx, dy } = DELTA[dir];
  return { x: x + dx, y: y + dy };
}

export function inBounds(floor: FloorState, x: number, y: number): boolean {
  return x >= 0 && y >= 0 && x < floor.width && y < floor.height;
}

export function getTile(floor: FloorState, x: number, y: number): Tile {
  return floor.tiles[y][x];
}

export function cloneTiles(tiles: Tile[][]): Tile[][] {
  return tiles.map((row) => row.map((t) => ({ ...t })));
}

export function cloneFloor(floor: FloorState): FloorState {
  return {
    ...floor,
    tiles: cloneTiles(floor.tiles),
    upTarget: floor.upTarget ? { ...floor.upTarget } : undefined,
    downTarget: floor.downTarget ? { ...floor.downTarget } : undefined,
  };
}

export function cloneGameState(state: GameState): GameState {
  const floors: GameState['floors'] = {};
  for (const id of Object.keys(state.floors)) {
    floors[id] = cloneFloor(state.floors[id]);
  }
  return {
    ...state,
    hero: { ...state.hero },
    floors,
    activeShopId: state.activeShopId,
  };
}

export function setTile(floor: FloorState, x: number, y: number, tile: Tile): void {
  floor.tiles[y][x] = tile;
}

Create `src/core/step.ts`（本文件一次写全移动/门/拾取/楼梯/战斗/商店，后续 Task 只加测试）:

```ts
import { canDefeat, forecastDamage } from './combat';
import { cloneGameState, getTile, inBounds, neighbor, setTile } from './map';
import type {
  Command,
  DataPack,
  GameState,
  MonsterDef,
  ReasonCode,
  StepResult,
  Tile,
  TileKind,
} from './types';

function reject(state: GameState, reason: ReasonCode): StepResult {
  const next = cloneGameState(state);
  next.lastReason = reason;
  return { state: next, reason };
}

function accept(state: GameState, reason: ReasonCode = 'ok'): StepResult {
  state.lastReason = reason;
  return { state, reason };
}

function doorKeyField(
  kind: TileKind,
): 'yellowKeys' | 'blueKeys' | 'redKeys' | null {
  if (kind === 'yellowDoor') return 'yellowKeys';
  if (kind === 'blueDoor') return 'blueKeys';
  if (kind === 'redDoor') return 'redKeys';
  return null;
}

function keyField(
  kind: TileKind,
): 'yellowKeys' | 'blueKeys' | 'redKeys' | null {
  if (kind === 'yellowKey') return 'yellowKeys';
  if (kind === 'blueKey') return 'blueKeys';
  if (kind === 'redKey') return 'redKeys';
  return null;
}

function findMonster(pack: DataPack, id: string): MonsterDef {
  const m = pack.monsters.find((x) => x.id === id);
  if (!m) throw new Error(`unknown monster: ${id}`);
  return m;
}

function applyPickup(state: GameState, tile: Tile, pack: DataPack): void {
  const kf = keyField(tile.kind);
  if (kf) {
    state.hero[kf] += 1;
    return;
  }
  if (tile.kind === 'redPotion') {
    const item = pack.items.find((i) => i.kind === 'redPotion');
    state.hero.hp += item?.hp ?? 50;
    return;
  }
  if (tile.kind === 'bluePotion') {
    const item = pack.items.find((i) => i.kind === 'bluePotion');
    state.hero.hp += item?.hp ?? 200;
    return;
  }
  if (tile.kind === 'redGem') {
    const item = pack.items.find((i) => i.kind === 'redGem');
    state.hero.atk += item?.atk ?? 1;
    return;
  }
  if (tile.kind === 'blueGem') {
    const item = pack.items.find((i) => i.kind === 'blueGem');
    state.hero.def += item?.def ?? 1;
  }
}

function handleMove(
  state: GameState,
  dir: Extract<Command, { type: 'Move' }>,
  pack: DataPack,
): StepResult {
  if (state.mode !== 'playing') return reject(state, 'noop');

  const floor = state.floors[state.hero.floorId];
  const { x, y } = neighbor(state.hero.x, state.hero.y, dir.dir);
  if (!inBounds(floor, x, y)) return reject(state, 'out_of_bounds');

  const next = cloneGameState(state);
  const f = next.floors[next.hero.floorId];
  const dest = getTile(f, x, y);

  if (dest.kind === 'wall') return reject(state, 'blocked_wall');

  const dk = doorKeyField(dest.kind);
  if (dk) {
    if (next.hero[dk] < 1) return reject(state, 'blocked_door');
    next.hero[dk] -= 1;
    setTile(f, x, y, { kind: 'floor' });
    next.hero.x = x;
    next.hero.y = y;
    return accept(next);
  }

  if (dest.kind === 'monster') {
    const monster = findMonster(pack, dest.ref ?? '');
    if (!canDefeat(next.hero, monster)) return reject(state, 'blocked_monster');
    const damage = forecastDamage(next.hero, monster)!;
    next.hero.hp -= damage;
    next.hero.gold += monster.gold;
    setTile(f, x, y, { kind: 'floor' });
    next.hero.x = x;
    next.hero.y = y;
    return accept(next);
  }

  if (dest.kind === 'shop') {
    next.hero.x = x;
    next.hero.y = y;
    next.mode = 'shop';
    next.activeShopId = dest.ref;
    return accept(next);
  }

  if (dest.kind === 'upStairs' && f.upTarget) {
    next.hero.floorId = f.upTarget.floorId;
    next.hero.x = f.upTarget.x;
    next.hero.y = f.upTarget.y;
    return accept(next);
  }

  if (dest.kind === 'downStairs' && f.downTarget) {
    next.hero.floorId = f.downTarget.floorId;
    next.hero.x = f.downTarget.x;
    next.hero.y = f.downTarget.y;
    return accept(next);
  }

  const pickupKinds: TileKind[] = [
    'yellowKey',
    'blueKey',
    'redKey',
    'redPotion',
    'bluePotion',
    'redGem',
    'blueGem',
  ];
  if (pickupKinds.includes(dest.kind)) {
    applyPickup(next, dest, pack);
    setTile(f, x, y, { kind: 'floor' });
    next.hero.x = x;
    next.hero.y = y;
    return accept(next);
  }

  next.hero.x = x;
  next.hero.y = y;
  return accept(next);
}

function handleBuy(state: GameState, optionIndex: number, pack: DataPack): StepResult {
  if (state.mode !== 'shop' || !state.activeShopId) return reject(state, 'noop');
  const shop = pack.shops.find((s) => s.id === state.activeShopId);
  if (!shop) return reject(state, 'noop');
  const option = shop.options[optionIndex];
  if (!option) return reject(state, 'noop');
  if (state.hero.gold < option.cost) return reject(state, 'cannot_afford');

  const next = cloneGameState(state);
  next.hero.gold -= option.cost;
  next.hero.hp += option.hp ?? 0;
  next.hero.atk += option.atk ?? 0;
  next.hero.def += option.def ?? 0;
  return accept(next, 'bought');
}

export function step(state: GameState, command: Command, pack: DataPack): StepResult {
  switch (command.type) {
    case 'Move':
      return handleMove(state, command, pack);
    case 'Buy':
      return handleBuy(state, command.optionIndex, pack);
    case 'Cancel': {
      if (state.mode !== 'shop') return reject(state, 'noop');
      const next = cloneGameState(state);
      next.mode = 'playing';
      next.activeShopId = undefined;
      return accept(next, 'left_shop');
    }
    case 'Confirm':
      return reject(state, 'noop');
    default:
      return reject(state, 'noop');
  }
}

import type { GameState } from './types';

function isGameState(value: unknown): value is GameState {
  if (!value || typeof value !== 'object') return false;
  const v = value as Record<string, unknown>;
  if (!v.hero || typeof v.hero !== 'object') return false;
  if (!v.floors || typeof v.floors !== 'object') return false;
  if (v.mode !== 'playing' && v.mode !== 'shop') return false;
  const hero = v.hero as Record<string, unknown>;
  for (const key of [
    'floorId',
    'x',
    'y',
    'hp',
    'atk',
    'def',
    'gold',
    'yellowKeys',
    'blueKeys',
    'redKeys',
  ]) {
    if (!(key in hero)) return false;
  }
  return true;
}

export function serialize(state: GameState): string {
  return JSON.stringify(state);
}

export function deserialize(json: string): GameState | null {
  try {
    const parsed: unknown = JSON.parse(json);
    if (!isGameState(parsed)) return null;
    return parsed;
  } catch {
    return null;
  }
}

import type { DataPack, FloorData, FloorState, GameState, Tile, TileKind } from './types';

const KIND_SET = new Set<string>([
  'floor',
  'wall',
  'upStairs',
  'downStairs',
  'yellowDoor',
  'blueDoor',
  'redDoor',
  'yellowKey',
  'blueKey',
  'redKey',
  'redPotion',
  'bluePotion',
  'redGem',
  'blueGem',
  'monster',
  'shop',
]);

export function parseTileToken(token: string): Tile {
  if (token.startsWith('monster:')) {
    return { kind: 'monster', ref: token.slice('monster:'.length) };
  }
  if (token.startsWith('shop:')) {
    return { kind: 'shop', ref: token.slice('shop:'.length) };
  }
  if (!KIND_SET.has(token)) {
    throw new Error(`unknown tile token: ${token}`);
  }
  return { kind: token as TileKind };
}

export function floorDataToState(data: FloorData): FloorState {
  if (data.grid.length !== data.height) {
    throw new Error(`floor ${data.id}: height mismatch`);
  }
  const tiles = data.grid.map((row) => {
    if (row.length !== data.width) {
      throw new Error(`floor ${data.id}: width mismatch`);
    }
    return row.map(parseTileToken);
  });
  return {
    id: data.id,
    width: data.width,
    height: data.height,
    tiles,
    upTarget: data.upTarget ? { ...data.upTarget } : undefined,
    downTarget: data.downTarget ? { ...data.downTarget } : undefined,
  };
}

export function createInitialState(pack: DataPack): GameState {
  const floors: GameState['floors'] = {};
  for (const f of pack.floors) {
    floors[f.id] = floorDataToState(f);
  }
  const s = pack.start;
  return {
    hero: {
      floorId: s.floorId,
      x: s.x,
      y: s.y,
      hp: s.hp,
      atk: s.atk,
      def: s.def,
      gold: s.gold,
      yellowKeys: 0,
      blueKeys: 0,
      redKeys: 0,
    },
    floors,
    mode: 'playing',
    lastReason: 'ok',
  };
}

import type { Command, Dir, GameMode } from '../../core/types';

export type SaveLoadHandler = {
  onSave: (slot: number) => void;
  onLoad: (slot: number) => void;
};

export function bindInput(
  target: Window,
  getMode: () => GameMode,
  onCommand: (command: Command) => void,
  saveLoad: SaveLoadHandler,
): () => void {
  const keyToDir: Record<string, Dir> = {
    ArrowUp: 'up',
    ArrowDown: 'down',
    ArrowLeft: 'left',
    ArrowRight: 'right',
  };

  function onKeyDown(ev: KeyboardEvent): void {
    const dir = keyToDir[ev.key];
    if (dir) {
      ev.preventDefault();
      if (getMode() === 'playing') onCommand({ type: 'Move', dir });
      return;
    }
    if (ev.key === 'Escape') {
      onCommand({ type: 'Cancel' });
      return;
    }
    if (ev.key >= '1' && ev.key <= '3') {
      const slot = Number(ev.key);
      if (ev.ctrlKey || ev.metaKey) {
        ev.preventDefault();
        saveLoad.onSave(slot);
        return;
      }
      if (ev.altKey) {
        ev.preventDefault();
        saveLoad.onLoad(slot);
        return;
      }
      if (getMode() === 'shop') {
        onCommand({ type: 'Buy', optionIndex: slot - 1 });
      }
      return;
    }
    if (ev.key >= '4' && ev.key <= '9' && getMode() === 'shop') {
      onCommand({ type: 'Buy', optionIndex: Number(ev.key) - 1 });
    }
  }

  target.addEventListener('keydown', onKeyDown);
  return () => target.removeEventListener('keydown', onKeyDown);
}

import packJson from '../../data/easy/pack.json';
import {
  createInitialState,
  deserialize,
  serialize,
  step,
} from '../../core';
import type { Command, DataPack, GameState } from '../../core/types';
import { bindInput } from './input';
import { mountPanel } from './panel';
import { drawGame, resizeCanvas } from './render';
import { loadSlot, saveSlot } from './storage';

const pack = packJson as DataPack;

function boot(): void {
  const app = document.getElementById('app');
  if (!app) throw new Error('#app missing');

  app.innerHTML = `
    <div id="layout">
      <canvas id="view"></canvas>
      <aside id="side"></aside>
    </div>
  `;
  const canvas = document.getElementById('view') as HTMLCanvasElement;
  const side = document.getElementById('side') as HTMLElement;
  const ctx = canvas.getContext('2d');
  if (!ctx) throw new Error('2d context unavailable');

  const panel = mountPanel(side);
  let state: GameState = createInitialState(pack);

  function paint(): void {
    const floor = state.floors[state.hero.floorId];
    resizeCanvas(canvas, floor);
    drawGame(ctx, state);
    panel.update(state, pack);
  }

  function apply(command: Command): void {
    const result = step(state, command, pack);
    state = result.state;
    panel.toast(result.reason);
    paint();
  }

  bindInput(
    window,
    () => state.mode,
    apply,
    {
      onSave(slot) {
        saveSlot(slot, serialize(state));
        panel.notify(`已存档 ${slot}`);
      },
      onLoad(slot) {
        const raw = loadSlot(slot);
        if (!raw) {
          panel.notify(`槽 ${slot} 空`);
          return;
        }
        const loaded = deserialize(raw);
        if (!loaded) {
          panel.notify(`槽 ${slot} 损坏`);
          return;
        }
        state = loaded;
        paint();
        panel.notify(`已读档 ${slot}`);
      },
    },
  );

  paint();
}

boot();
```
